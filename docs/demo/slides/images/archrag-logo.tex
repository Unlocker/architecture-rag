\documentclass[tikz,border=0pt]{standalone}
\usepackage[T2A]{fontenc}
\usepackage[utf8]{inputenc}
\usetikzlibrary{shapes.geometric,calc}
\definecolor{darkred}{rgb}{0.8,0,0}      % beaver: titlelike fg
\definecolor{beaverred}{rgb}{0.6,0,0}    % тёмный акцент
\definecolor{beavergray}{gray}{0.35}
\begin{document}
\begin{tikzpicture}[x=1mm,y=1mm,line width=0.55pt]
  % холст 64.5 x 20 мм  (3.23:1)
  \path[use as bounding box] (0,0) rectangle (64.5,20);

  % --- база данных (цилиндр) ---
  \node[cylinder, shape border rotate=90, draw=beaverred, fill=darkred!12,
        minimum width=6.2mm, minimum height=8.2mm, aspect=0.32,
        line width=0.6pt] (db) at (5,10) {};

  % --- очередь сообщений (сегменты) ---

    \draw[draw=beavergray, fill=gray!15] (11.2+\i*2.4,7.6) rectangle ++(1.9,4.8);

  \draw[beavergray, line width=1.3pt] (27.2,10) circle (4.9);
  \draw[white, line width=0.8pt] (24.3,12.4) arc[start angle=140,end angle=112,radius=4.3];
  \draw[beavergray, line width=2.6pt, line cap=round] ($(27.2,10)+(-45:4.9)+(-45:0.45)$) -- ($(27.2,10)+(-45:8.2)$);

  % --- надпись ---
  \node[anchor=west, inner sep=0pt] at (34.6,10)
    {\fontsize{15.5}{15.5}\selectfont\sffamily\bfseries{\color{darkred}Arch}\,{\color{beavergray}RAG}};
